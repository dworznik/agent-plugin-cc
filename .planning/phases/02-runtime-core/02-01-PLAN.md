import { existsSync, readFileSync, accessSync, constants } from 'node:fs';
import { dirname, join } from 'node:path';
import { AgentsConfigSchema } from './schema.js';
import type { AgentsConfig } from '../types.js';

export class ConfigNotFoundError extends Error {
  constructor(startDir: string) {
    super(`agents.json not found: walked up from ${startDir} to filesystem root without finding .claude/agents.json`);
    this.name = 'ConfigNotFoundError';
  }
}

export class ConfigValidationError extends Error {
  public readonly fieldPath: string;
  constructor(fieldPath: string, message: string) {
    super(`agents.json invalid at '${fieldPath}': ${message}`);
    this.name = 'ConfigValidationError';
    this.fieldPath = fieldPath;
  }
}

export class PluginDataError extends Error {
  constructor(reason: string) {
    super(`CLAUDE_PLUGIN_DATA ${reason}`);
    this.name = 'PluginDataError';
  }
}

// Walk up from startDir to find .claude/agents.json (per CONF-04, D-06)
export function findConfigPath(startDir: string): string {
  let dir = startDir;
  while (true) {
    const candidate = join(dir, '.claude', 'agents.json');
    if (existsSync(candidate)) return candidate;
    const parent = dirname(dir);
    if (parent === dir) throw new ConfigNotFoundError(startDir);
    dir = parent;
  }
}

// Load and validate agents.json (per CONF-02). Returns typed config.
export function loadConfig(cwd: string): AgentsConfig {
  const configPath = findConfigPath(cwd);
  let raw: unknown;
  try {
    raw = JSON.parse(readFileSync(configPath, 'utf-8'));
  } catch {
    throw new ConfigValidationError('(root)', 'File is not valid JSON');
  }
  const result = AgentsConfigSchema.safeParse(raw);
  if (!result.success) {
    const issue = result.error.errors[0];
    const fieldPath = issue.path.join('.') || '(root)';
    throw new ConfigValidationError(fieldPath, issue.message);
  }
  return result.data;
}

// Validate CLAUDE_PLUGIN_DATA env var is set and writable (per ERR-05)
export function validatePluginDataDir(): string {
  const dir = process.env['CLAUDE_PLUGIN_DATA'];
  if (!dir) {
    throw new PluginDataError('environment variable is not set. Claude Code must set CLAUDE_PLUGIN_DATA before invoking the companion.');
  }
  try {
    accessSync(dir, constants.W_OK);
  } catch {
    throw new PluginDataError(`directory is not writable: ${dir}`);
  }
  return dir;
}

```typescript
import { spawn } from 'node:child_process';
import type { ProcessResult } from '../types.js';

export class AdapterTimeoutError extends Error {
  public readonly timeoutMs: number;
  constructor(timeoutMs: number) {
    super(`Adapter process timed out after ${timeoutMs}ms`);
    this.name = 'AdapterTimeoutError';
    this.timeoutMs = timeoutMs;
  }
}

export class AdapterProcessError extends Error {
  public readonly exitCode: number | null;
  public readonly stderr: string;
  constructor(exitCode: number | null, stderr: string) {
    super(`Adapter process exited with code ${exitCode}: ${stderr.slice(0, 500)}`);
    this.name = 'AdapterProcessError';
    this.exitCode = exitCode;
    this.stderr = stderr;
  }
}

export interface SpawnOptions {
  signal?: AbortSignal;
  env?: NodeJS.ProcessEnv;
  cwd?: string;
  timeoutMs?: number;
}

// Spawn a child process with optional AbortController timeout (per ADPT-09)
// Uses spawn (not exec) to avoid buffering issues (anti-pattern from research)
export function spawnWithTimeout(
  command: string,
  args: string[],
  opts: SpawnOptions = {}
): Promise<ProcessResult> {
  return new Promise((resolve, reject) => {
    const ac = new AbortController();
    const signal = opts.signal ?? ac.signal;
    const timeoutMs = opts.timeoutMs ?? 60_000;

    const timer = setTimeout(() => ac.abort(), timeoutMs);

    const proc = spawn(command, args, {
      signal,
      env: opts.env ?? process.env,
      cwd: opts.cwd,
      stdio: ['ignore', 'pipe', 'pipe'],
    });

    const stdoutChunks: Buffer[] = [];
    const stderrChunks: Buffer[] = [];

    proc.stdout.on('data', (chunk: Buffer) => stdoutChunks.push(chunk));
    proc.stderr.on('data', (chunk: Buffer) => stderrChunks.push(chunk));

    proc.on('close', (code) => {
      clearTimeout(timer);
      resolve({
        stdout: Buffer.concat(stdoutChunks).toString('utf-8'),
        stderr: Buffer.concat(stderrChunks).toString('utf-8'),
        exitCode: code,
      });
    });

    proc.on('error', (err) => {
      clearTimeout(timer);
      if ((err as NodeJS.ErrnoException).code === 'ABORT_ERR') {
        reject(new AdapterTimeoutError(timeoutMs));
      } else {
        reject(err);
      }
    });
  });
}

// Safe JSON parse — returns parsed value or null on malformed input (per ADPT-10)
export function parseJSON<T = unknown>(str: string): T | null {
  try {
    return JSON.parse(str) as T;
  } catch {
    return null;
  }
}

// Safe JSONL parse — returns array of successfully parsed lines, skips invalid lines (per ADPT-10)
export function parseJSONL<T = unknown>(str: string): T[] {
  return str
    .split('\n')
    .filter((line) => line.trim().length > 0)
    .map((line) => {
      try {
        return JSON.parse(line) as T;
      } catch {
        return null;
      }
    })
    .filter((item): item is T => item !== null);
}
```
