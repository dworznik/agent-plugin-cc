```typescript
import { createHash } from 'node:crypto';

// SHA-256 hash of absolute workspace root path for deterministic partitioning (per D-08, SESS-04)
export function workspaceHash(absolutePath: string): string {
  return createHash('sha256').update(absolutePath).digest('hex');
}
```

```typescript
import { existsSync, readFileSync, writeFileSync, unlinkSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';
import { getSessionPath } from './paths.js';
import type { SessionData } from '../types.js';

// Load session from disk (per SESS-02). Returns null if not found.
// Corrupt files trigger warning + deletion per D-09.
export function loadSession(
  pluginDataDir: string,
  workspaceRoot: string,
  agentId: string,
  warn?: (msg: string) => void
): SessionData | null {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  if (!existsSync(path)) return null;
  try {
    const raw = readFileSync(path, 'utf-8');
    return JSON.parse(raw) as SessionData;
  } catch {
    // Corrupt session: warn + delete + fresh start (per D-09)
    warn?.(`Corrupt session file at ${path}, deleting and starting fresh`);
    try { unlinkSync(path); } catch { /* ignore if already gone */ }
    return null;
  }
}

// Save session to disk (per SESS-01). Creates directories recursively.
// Uses flag:'wx' for first write to avoid race condition (research Pitfall 3).
export function saveSession(
  pluginDataDir: string,
  workspaceRoot: string,
  agentId: string,
  session: SessionData
): void {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  mkdirSync(dirname(path), { recursive: true });
  writeFileSync(path, JSON.stringify(session, null, 2), 'utf-8');
}

// Delete session file (per SESS-03 --fresh).
export function deleteSession(
  pluginDataDir: string,
  workspaceRoot: string,
  agentId: string
): void {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  if (existsSync(path)) {
    unlinkSync(path);
  }
}
```

All functions accept `pluginDataDir` as parameter (not read from env) so runtime stays a library. Companion passes the validated env var.

Create `packages/runtime/src/logging/logger.ts` (per LOG-01, LOG-02, LOG-03):

```typescript
import { appendFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';
import type { LogEntry, HookSet } from '../types.js';

export interface Logger {
  info(message: string, durationMs?: number): void;
  warn(message: string): void;
  error(message: string): void;
}

// Create a logger that writes structured JSON log entries to a file (per LOG-01)
// and calls emitLog hook for each entry (per LOG-03)
export function createLogger(
  jobId: string,
  logPath: string,
  emitLog: HookSet['emitLog']
): Logger {
  // Ensure log directory exists
  mkdirSync(dirname(logPath), { recursive: true });

  function write(level: LogEntry['level'], message: string, durationMs?: number): void {
    const entry: LogEntry = {
      timestamp: new Date().toISOString(),
      level,
      jobId,
      message,
      ...(durationMs !== undefined ? { durationMs } : {}),
    };
    // Write to file (per LOG-01)
    appendFileSync(logPath, JSON.stringify(entry) + '\n', 'utf-8');
    // Call emitLog hook (per LOG-03)
    emitLog(entry);
  }

  return {
    info: (message: string, durationMs?: number) => write('info', message, durationMs),
    warn: (message: string) => write('warn', message),
    error: (message: string) => write('error', message),
  };
}
```

```typescript
import type { HookSet, SessionData, LogEntry } from '../types.js';
import { loadSession as storeLoadSession, saveSession as storeSaveSession } from '../session/store.js';
import { validatePluginDataDir } from '../config/loader.js';

// Default hook implementations (per D-03)
// Each hook has a sensible default. Companion can override any hook.
export const defaultHooks: HookSet = {
  // Default: pass through base env unchanged
  buildAgentEnv: (_agentId: string, baseEnv: Record<string, string>): Record<string, string> => baseEnv,

  // Default: pass through prompt unchanged
  resolveInstructions: (_agentId: string, prompt: string): string => prompt,

  // Default: load from CLAUDE_PLUGIN_DATA session store
  loadSession: (workspaceRoot: string, agentId: string): SessionData | null => {
    const pluginDataDir = validatePluginDataDir();
    return storeLoadSession(pluginDataDir, workspaceRoot, agentId, console.warn);
  },

  // Default: save to CLAUDE_PLUGIN_DATA session store
  saveSession: (workspaceRoot: string, agentId: string, session: SessionData): void => {
    const pluginDataDir = validatePluginDataDir();
    storeSaveSession(pluginDataDir, workspaceRoot, agentId, session);
  },

  // Default: no-op (logs are written to file by logger, this hook is for interception)
  emitLog: (_entry: LogEntry): void => {},
};
```

The `defaultHooks` object satisfies the `HookSet` interface exactly. Each default is minimal: passthrough for env/instructions, store-backed for session, no-op for logging interception. LOG-02 (execution timing) is handled by the adapter calling `logger.info('Adapter execution started')` and `logger.info('Adapter execution completed', durationMs)` -- the logger records timestamps on every entry.
