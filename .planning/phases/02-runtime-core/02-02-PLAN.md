import { createHash } from 'node:crypto';

// SHA-256 hash of absolute workspace root path for deterministic partitioning (per D-08, SESS-04)
export function workspaceHash(absolutePath: string): string {
  return createHash('sha256').update(absolutePath).digest('hex');
}

```typescript
import { existsSync, readFileSync, writeFileSync, unlinkSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';
import { getSessionPath } from './paths.js';
import type { SessionData } from '../types.js';

// Load session from disk (per SESS-02). Returns null if not found.
// Corrupt files trigger warning + deletion per D-09.
export function loadSession(
  pluginDataDir: string, workspaceRoot: string, agentId: string,
  warn?: (msg: string) => void
): SessionData | null {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  if (!existsSync(path)) return null;
  try {
    const raw = readFileSync(path, 'utf-8');
    return JSON.parse(raw) as SessionData;
  } catch {
    warn?.(`Corrupt session file at ${path}, deleting and starting fresh`);
    try { unlinkSync(path); } catch { /* ignore if already gone */ }
    return null;
  }
}

// Save session to disk (per SESS-01). Creates directories recursively.
export function saveSession(
  pluginDataDir: string, workspaceRoot: string, agentId: string, session: SessionData
): void {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  mkdirSync(dirname(path), { recursive: true });
  writeFileSync(path, JSON.stringify(session, null, 2), 'utf-8');
}

// Delete session file (per SESS-03 --fresh).
export function deleteSession(
  pluginDataDir: string, workspaceRoot: string, agentId: string
): void {
  const path = getSessionPath(pluginDataDir, workspaceRoot, agentId);
  if (existsSync(path)) unlinkSync(path);
}
```

```typescript
import { appendFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';
import type { LogEntry, HookSet } from '../types.js';

export interface Logger {
  info(message: string, durationMs?: number): void;
  warn(message: string): void;
  error(message: string): void;
}

// Create a logger that writes structured JSON log entries to a file (per LOG-01)
// and calls emitLog hook for each entry (per LOG-03).
// durationMs parameter on info() supports LOG-02 execution timing.
export function createLogger(
  jobId: string, logPath: string, emitLog: HookSet['emitLog']
): Logger {
  mkdirSync(dirname(logPath), { recursive: true });

  function write(level: LogEntry['level'], message: string, durationMs?: number): void {
    const entry: LogEntry = {
      timestamp: new Date().toISOString(),
      level,
      jobId,
      message,
      ...(durationMs !== undefined ? { durationMs } : {}),
    };
    appendFileSync(logPath, JSON.stringify(entry) + '\n', 'utf-8');
    emitLog(entry);
  }

  return {
    info: (message: string, durationMs?: number) => write('info', message, durationMs),
    warn: (message: string) => write('warn', message),
    error: (message: string) => write('error', message),
  };
}
```
