```typescript
import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'node:fs';
import { join, dirname } from 'node:path';

export type JobStatus = 'pending' | 'running' | 'completed' | 'failed' | 'cancelled';

export interface JobMetadata {
  jobId: string;
  agentId: string;
  status: JobStatus;
  startTime: string;      // ISO timestamp
  endTime?: string;        // ISO timestamp, set on terminal state
  pid: number | null;      // null until worker starts
  error?: string;          // set on 'failed' status (ERR-04)
}

export function getJobPath(statePath: string, jobId: string): string {
  return join(statePath, 'jobs', `${jobId}.json`);
}

export function getResultPath(statePath: string, jobId: string): string {
  return join(statePath, 'results', jobId);
}

export function getCancelPath(statePath: string, jobId: string): string {
  return join(statePath, 'cancel', jobId);
}

export function readJob(statePath: string, jobId: string): JobMetadata | null {
  const p = getJobPath(statePath, jobId);
  if (!existsSync(p)) return null;
  return JSON.parse(readFileSync(p, 'utf-8')) as JobMetadata;
}

export function writeJob(statePath: string, meta: JobMetadata): void {
  const p = getJobPath(statePath, meta.jobId);
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, JSON.stringify(meta, null, 2), 'utf-8');
}

export function updateJob(statePath: string, jobId: string, update: Partial<JobMetadata>): void {
  const existing = readJob(statePath, jobId);
  if (!existing) throw new Error(`Job not found: ${jobId}`);
  writeJob(statePath, { ...existing, ...update });
}
```

**packages/plugin/src/companion/commands/setup.ts** — Setup subcommand (per D-08, CONF-05, CONF-06):

The companion `setup` accepts `--config '<JSON>'` where Claude assembles the JSON from user conversation. The companion validates and writes it.

```typescript
import { writeFileSync, mkdirSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { AgentsConfigSchema } from '@agent-plugin-cc/runtime';

export async function setupCommand(args: string[]): Promise<void> {
  // Parse --config flag
  const configIdx = args.indexOf('--config');
  if (configIdx === -1 || configIdx + 1 >= args.length) {
    console.log(JSON.stringify({ error: 'Usage: setup --config \'<json>\'' }));
    process.exit(1);
  }
  const rawJson = args[configIdx + 1];

  let parsed: unknown;
  try {
    parsed = JSON.parse(rawJson);
  } catch {
    console.log(JSON.stringify({ error: 'Invalid JSON in --config argument' }));
    process.exit(1);
  }

  // Validate against schema
  const result = AgentsConfigSchema.safeParse(parsed);
  if (!result.success) {
    const issue = result.error.errors[0];
    const fieldPath = issue.path.join('.') || '(root)';
    console.log(JSON.stringify({ error: `Validation failed at '${fieldPath}': ${issue.message}` }));
    process.exit(1);
  }

  // Write to .claude/agents.json relative to cwd
  const outPath = join(process.cwd(), '.claude', 'agents.json');
  mkdirSync(dirname(outPath), { recursive: true });
  writeFileSync(outPath, JSON.stringify(result.data, null, 2), 'utf-8');

  console.log(JSON.stringify({ success: true, path: outPath, agents: result.data.agents.length }));
}
```

```typescript
import { loadConfig, ConfigNotFoundError, ConfigValidationError } from '@agent-plugin-cc/runtime';

export async function listCommand(_args: string[]): Promise<void> {
  let config;
  try {
    config = loadConfig(process.cwd());
  } catch (err) {
    if (err instanceof ConfigNotFoundError) {
      console.log(JSON.stringify({ error: 'No agents.json found. Run /agents:setup to configure agents.' }));
      process.exit(1);
    }
    if (err instanceof ConfigValidationError) {
      console.log(JSON.stringify({ error: err.message }));
      process.exit(1);
    }
    throw err;
  }

  const agents = config.agents.map(a => ({
    id: a.id,
    label: a.label,
    description: a.description,
    adapterType: a.adapterType,
  }));

  console.log(JSON.stringify({ agents, defaultAgent: config.defaultAgent ?? null }));
}
```

**packages/plugin/tests/helpers/companion-runner.ts** — Helper to run the bundled companion as a child process:

```typescript
import { execFile } from 'node:child_process';
import { join } from 'node:path';

const COMPANION_PATH = join(__dirname, '..', '..', 'scripts', 'agents-companion.cjs');

export interface CompanionResult {
  stdout: string;
  stderr: string;
  exitCode: number;
  json: Record<string, unknown> | null;
}

export function runCompanion(
  args: string[],
  opts?: { cwd?: string; env?: Record<string, string> }
): Promise<CompanionResult> {
  return new Promise((resolve) => {
    execFile('node', [COMPANION_PATH, ...args], {
      cwd: opts?.cwd ?? process.cwd(),
      env: { ...process.env, ...opts?.env },
      timeout: 30_000,
    }, (error, stdout, stderr) => {
      const exitCode = error?.code ? Number(error.code) : (error ? 1 : 0);
      let json: Record<string, unknown> | null = null;
      try { json = JSON.parse(stdout.trim()); } catch { /* not JSON */ }
      resolve({ stdout: stdout.trim(), stderr: stderr.trim(), exitCode, json });
    });
  });
}
```

**packages/plugin/tests/helpers/temp-env.ts** — Helper to create isolated temp directories for each test:

```typescript
import { mkdirSync, writeFileSync, rmSync, mkdtempSync } from 'node:fs';
import { join } from 'node:path';
import { tmpdir } from 'node:os';
import { afterEach, beforeEach } from 'vitest';

export interface TestEnv {
  tmpDir: string;
  pluginDataDir: string;
  workspaceDir: string;
}

let currentEnv: TestEnv | null = null;

export function useTempEnv(): { getEnv: () => TestEnv } {
  beforeEach(() => {
    const tmpDir = mkdtempSync(join(tmpdir(), 'companion-test-'));
    const pluginDataDir = join(tmpDir, 'plugin-data');
    const workspaceDir = join(tmpDir, 'workspace');
    mkdirSync(pluginDataDir, { recursive: true });
    mkdirSync(workspaceDir, { recursive: true });
    currentEnv = { tmpDir, pluginDataDir, workspaceDir };
  });

  afterEach(() => {
    if (currentEnv) {
      rmSync(currentEnv.tmpDir, { recursive: true, force: true });
      currentEnv = null;
    }
  });

  return {
    getEnv: () => {
      if (!currentEnv) throw new Error('Test env not initialized');
      return currentEnv;
    }
  };
}

export function writeAgentsConfig(workspaceDir: string, config: Record<string, unknown>): void {
  const dir = join(workspaceDir, '.claude');
  mkdirSync(dir, { recursive: true });
  writeFileSync(join(dir, 'agents.json'), JSON.stringify(config, null, 2), 'utf-8');
}
```
