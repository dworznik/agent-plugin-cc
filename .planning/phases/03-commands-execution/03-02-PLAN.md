```typescript
import { readFileSync, existsSync } from 'node:fs';

export async function resultCommand(args: string[]): Promise<void> {
  // Config guard (ERR-01, ERR-02)
  // ...

  const jobId = args[0];
  if (!jobId) {
    console.log(JSON.stringify({ error: 'Usage: result <job-id>' }));
    process.exit(1);
  }

  const pluginDataDir = validatePluginDataDir();
  const statePath = getStatePath(pluginDataDir, process.cwd());
  const job = readJob(statePath, jobId);

  if (!job) {
    console.log(JSON.stringify({ error: `Job not found: ${jobId}` }));
    process.exit(1);
  }

  if (job.status !== 'completed') {
    // CMD-11: error if not completed
    if (job.status === 'failed') {
      console.log(JSON.stringify({ error: `Job failed: ${job.error ?? 'unknown error'}` }));
    } else {
      console.log(JSON.stringify({ error: `Job not completed. Current status: ${job.status}` }));
    }
    process.exit(1);
  }

  const resultPath = getResultPath(statePath, jobId);
  if (!existsSync(resultPath)) {
    console.log(JSON.stringify({ error: `Result file missing for job: ${jobId}` }));
    process.exit(1);
  }

  const output = readFileSync(resultPath, 'utf-8');
  console.log(JSON.stringify({ jobId, output }));
}
```

```typescript
import { writeFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';

export async function cancelCommand(args: string[]): Promise<void> {
  // Config guard (ERR-01, ERR-02)
  // ...

  const jobId = args[0];
  if (!jobId) {
    console.log(JSON.stringify({ error: 'Usage: cancel <job-id>' }));
    process.exit(1);
  }

  const pluginDataDir = validatePluginDataDir();
  const statePath = getStatePath(pluginDataDir, process.cwd());
  const job = readJob(statePath, jobId);

  if (!job) {
    console.log(JSON.stringify({ error: `Job not found: ${jobId}` }));
    process.exit(1);
  }

  if (job.status !== 'running' && job.status !== 'pending') {
    // CMD-13: error if not in cancellable state
    console.log(JSON.stringify({ error: `Cannot cancel job in '${job.status}' state. Only 'running' or 'pending' jobs can be cancelled.` }));
    process.exit(1);
  }

  // Write cancel sentinel (JOB-06)
  const cancelPath = getCancelPath(statePath, jobId);
  mkdirSync(dirname(cancelPath), { recursive: true });
  writeFileSync(cancelPath, '', 'utf-8');

  // Update job metadata to cancelled
  updateJob(statePath, jobId, { status: 'cancelled', endTime: new Date().toISOString() });

  console.log(JSON.stringify({ jobId, status: 'cancelled' }));
}
```
